1//! # Intel (Vulkan) IVF Index
2//!
3//! Mirror of `CudaIvfIndex` / `RocmIvfIndex` (see the `cuda` / `rocm`
4//! features) on the Vulkan Compute backend. Uses the pre-built `sgemv_dot`
5//! and `sgemm_dot` compute pipelines for every GPU-side dot-product pass;
6//! argmin and top-K run on the host, matching the CUDA/ROCm/Metal scope
7//! of v1.
8//!
9//! ⚠️ AUTHORED BLIND — see `phase3c_add-intel-backend`.
10

116    pub fn build(&mut self, vectors: &[GpuVector]) -> Result<()> {
117        if vectors.is_empty() {
118            return Err(HiveGpuError::InvalidConfiguration(
119                "cannot build IVF from empty vector set".to_string(),
120            ));
121        }
122        if vectors.len() < self.config.n_list {
123            return Err(HiveGpuError::InvalidConfiguration(format!(
124                "need at least n_list={} vectors to train, got {}",
125                self.config.n_list,
126                vectors.len()
127            )));
128        }
129        for (i, v) in vectors.iter().enumerate() {
130            if v.data.len() != self.dimension {
131                return Err(HiveGpuError::DimensionMismatch {
132                    expected: self.dimension,
133                    actual: v.data.len(),
134                });
135            }
136            if v.data.iter().any(|x| !x.is_finite()) {
137                return Err(HiveGpuError::InvalidConfiguration(format!(
138                    "non-finite component in input vector #{i} (id={})",
139                    v.id
140                )));
141            }
142        }
143
144        let sample_size = self.config.training_sample_size.min(vectors.len());
145        let flat_sample: Vec<f32> = vectors
146            .iter()
147            .take(sample_size)
148            .flat_map(|v| v.data.iter().copied())
149            .collect();
150
151        info!(
152            "intel ivf build: dim={} n={} n_list={} training_sample={}",
153            self.dimension,
154            vectors.len(),
155            self.config.n_list,
156            sample_size
157        );
158
159        let centroids_flat =
160            self.train_kmeans(&flat_sample, sample_size, self.config.kmeans_iters)?;
161        debug_assert_eq!(centroids_flat.len(), self.config.n_list * self.dimension);
162
163        let flat_all: Vec<f32> = vectors
164            .iter()
165            .flat_map(|v| v.data.iter().copied())
166            .collect();
167        let assignments = self.assign_to_centroids(&flat_all, vectors.len(), &centroids_flat)?;
168        debug_assert_eq!(assignments.len(), vectors.len());
169
170        let (offsets, perm) = build_cluster_layout(&assignments, self.config.n_list);
171
172        let mut reordered = vec![0f32; flat_all.len()];
173        let mut reordered_ids = Vec::with_capacity(vectors.len());
174        let mut reordered_norms = Vec::with_capacity(vectors.len());
175        for (local_idx, &global_idx) in perm.iter().enumerate() {
176            let src = global_idx * self.dimension;
177            let dst = local_idx * self.dimension;
178            reordered[dst..dst + self.dimension]
179                .copy_from_slice(&flat_all[src..src + self.dimension]);
180            reordered_ids.push(vectors[global_idx].id.clone());
181            reordered_norms.push(dot_self(&flat_all[src..src + self.dimension]));
182        }
183
184        // Upload centroids.
185        let centroids_bytes = centroids_flat.len() * std::mem::size_of::<f32>();
186        let centroids_buf = allocate_host_visible_buffer(&self.context, centroids_bytes)?;
187        write_f32_slice(&centroids_buf, &centroids_flat)?;
188
189        // Upload reordered vectors.
190        let vectors_bytes = reordered.len() * std::mem::size_of::<f32>();
191        let vectors_buf = allocate_host_visible_buffer(&self.context, vectors_bytes)?;
192        write_f32_slice(&vectors_buf, &reordered)?;
193
194        let mut centroid_norms_sq = Vec::with_capacity(self.config.n_list);
195        for i in 0..self.config.n_list {
196            let start = i * self.dimension;
197            centroid_norms_sq.push(dot_self(&centroids_flat[start..start + self.dimension]));
198        }
199
200        if let Some(old) = self.centroids.take() {
201            old.destroy(self.context.device());
202        }
203        if let Some(old) = self.vectors.take() {
204            old.destroy(self.context.device());
205        }
206
207        self.centroids = Some(centroids_buf);
208        self.centroid_norms_sq = centroid_norms_sq;
209        self.vectors = Some(vectors_buf);
210        self.vector_norms_sq = reordered_norms;
211        self.cluster_offsets = offsets;
212        self.ids_by_local_index = reordered_ids;
213        self.vector_count = vectors.len();
214        self.trained = true;
215
216        info!(
217            "intel ivf build done: {} vectors across {} clusters",
218            self.vector_count, self.config.n_list
219        );
220        Ok(())
221    }

223    pub fn search(&self, query: &[f32], k: usize) -> Result<Vec<GpuSearchResult>> {
224        if !self.trained {
225            return Err(HiveGpuError::InvalidConfiguration(
226                "IVF index must be built before search".to_string(),
227            ));
228        }
229        if query.len() != self.dimension {
230            return Err(HiveGpuError::DimensionMismatch {
231                expected: self.dimension,
232                actual: query.len(),
233            });
234        }
235        if k == 0 || self.vector_count == 0 {
236            return Ok(Vec::new());
237        }
238        for (i, &x) in query.iter().enumerate() {
239            if !x.is_finite() {
240                return Err(HiveGpuError::InvalidConfiguration(format!(
241                    "non-finite query component at index {i}"
242                )));
243            }
244        }
245
246        let centroids = self
247            .centroids
248            .as_ref()
249            .ok_or_else(|| HiveGpuError::InvalidConfiguration("not trained".to_string()))?;
250        let vectors = self
251            .vectors
252            .as_ref()
253            .ok_or_else(|| HiveGpuError::InvalidConfiguration("not trained".to_string()))?;
254
255        // 1. Coarse: sgemv_dot against the centroids buffer.
256        let coarse = self.sgemv_dot_whole_buffer(centroids, self.config.n_list, query)?;
257        let query_norm_sq = dot_self(query);
258        let probe = select_nprobe_clusters(
259            &coarse,
260            &self.centroid_norms_sq,
261            query_norm_sq,
262            self.config.nprobe,
263        );
264
265        // 2. Refined: for each probed cluster, sgemv_dot against a
266        //    sub-range of the vectors buffer.
267        let query_bytes = query.len() * std::mem::size_of::<f32>();
268        let query_buf = allocate_host_visible_buffer(&self.context, query_bytes)?;
269        write_f32_slice(&query_buf, query)?;
270
271        let mut candidates: Vec<(usize, f32)> = Vec::new();
272        for cluster_id in &probe {
273            let start = self.cluster_offsets[*cluster_id];
274            let end = self.cluster_offsets[cluster_id + 1];
275            let count = end - start;
276            if count == 0 {
277                continue;
278            }
279
280            let scores_bytes = count * std::mem::size_of::<f32>();
281            let scores_buf = allocate_host_visible_buffer(&self.context, scores_bytes)?;
282
283            let matrix_byte_offset = (start * self.dimension * std::mem::size_of::<f32>()) as u64;
284            let matrix_byte_range = (count * self.dimension * std::mem::size_of::<f32>()) as u64;
285
286            let pipeline = self.context.sgemv_dot();
287            let pc = SgemvPushConstants {
288                dimension: self.dimension as u32,
289                n_vectors: count as u32,
290            };
291            let grid = (((count as u32) + 255) / 256, 1, 1);
292            dispatch_three_buffer_compute_ranged(
293                &self.context,
294                pipeline,
295                [
296                    (vectors, matrix_byte_offset, matrix_byte_range),
297                    (&query_buf, 0, ash::vk::WHOLE_SIZE),
298                    (&scores_buf, 0, ash::vk::WHOLE_SIZE),
299                ],
300                pc,
301                grid,
302            )?;
303
304            let scores = read_f32_vec(&scores_buf, count)?;
305            scores_buf.destroy(self.context.device());
306
307            for (j, dot) in scores.into_iter().enumerate() {
308                let local_idx = start + j;
309                let metric_score = self.score_from_dot(dot, local_idx, query_norm_sq);
310                candidates.push((local_idx, metric_score));
311            }
312        }
313
314        query_buf.destroy(self.context.device());
315
316        Ok(self.finalize_top_k(candidates, k))
317    }

321    fn sgemv_dot_whole_buffer(
322        &self,
323        matrix_buf: &VulkanBuffer,
324        n_rows: usize,
325        query: &[f32],
326    ) -> Result<Vec<f32>> {
327        let query_bytes = query.len() * std::mem::size_of::<f32>();
328        let query_buf = allocate_host_visible_buffer(&self.context, query_bytes)?;
329        write_f32_slice(&query_buf, query)?;
330        let scores_bytes = n_rows * std::mem::size_of::<f32>();
331        let scores_buf = allocate_host_visible_buffer(&self.context, scores_bytes)?;
332
333        let pipeline = self.context.sgemv_dot();
334        let pc = SgemvPushConstants {
335            dimension: self.dimension as u32,
336            n_vectors: n_rows as u32,
337        };
338        let grid = (((n_rows as u32) + 255) / 256, 1, 1);
339        dispatch_three_buffer_compute(
340            &self.context,
341            pipeline,
342            [matrix_buf, &query_buf, &scores_buf],
343            pc,
344            grid,
345        )?;
346
347        let scores = read_f32_vec(&scores_buf, n_rows)?;
348        query_buf.destroy(self.context.device());
349        scores_buf.destroy(self.context.device());
350        Ok(scores)
351    }

353    /// Run sgemm_dot over `(samples × centroids^T)` and reduce to
354    /// per-sample cluster ids on the host.
355    fn assign_to_centroids(
356        &self,
357        flat_samples: &[f32],
358        n_samples: usize,
359        centroids_flat: &[f32],
360    ) -> Result<Vec<u32>> {
361        let samples_bytes = flat_samples.len() * std::mem::size_of::<f32>();
362        let centroids_bytes = centroids_flat.len() * std::mem::size_of::<f32>();
363        let out_bytes = n_samples * self.config.n_list * std::mem::size_of::<f32>();
364
365        let samples_buf = allocate_host_visible_buffer(&self.context, samples_bytes)?;
366        write_f32_slice(&samples_buf, flat_samples)?;
367        let centroids_buf = allocate_host_visible_buffer(&self.context, centroids_bytes)?;
368        write_f32_slice(&centroids_buf, centroids_flat)?;
369        let out_buf = allocate_host_visible_buffer(&self.context, out_bytes)?;
370
371        let pipeline = self.context.sgemm_dot();
372        let pc = SgemmPushConstants {
373            dimension: self.dimension as u32,
374            n_list: self.config.n_list as u32,
375            n_samples: n_samples as u32,
376        };
377        let grid = (
378            ((n_samples as u32) + 15) / 16,
379            ((self.config.n_list as u32) + 15) / 16,
380            1,
381        );
382        dispatch_three_buffer_compute(
383            &self.context,
384            pipeline,
385            [&samples_buf, &centroids_buf, &out_buf],
386            pc,
387            grid,
388        )?;
389
390        let host_dots = read_f32_vec(&out_buf, n_samples * self.config.n_list)?;
391        samples_buf.destroy(self.context.device());
392        centroids_buf.destroy(self.context.device());
393        out_buf.destroy(self.context.device());
394
395        // Precompute centroid norms then argmax(2 * dot - ||c||^2).
396        let mut centroid_norms_sq = Vec::with_capacity(self.config.n_list);
397        for j in 0..self.config.n_list {
398            let start = j * self.dimension;
399            centroid_norms_sq.push(dot_self(&centroids_flat[start..start + self.dimension]));
400        }
401        let mut assignments = vec![0u32; n_samples];
402        for i in 0..n_samples {
403            let row = &host_dots[i * self.config.n_list..(i + 1) * self.config.n_list];
404            let (best_j, _) = row
405                .iter()
406                .enumerate()
407                .map(|(j, &dot)| (j, 2.0 * dot - centroid_norms_sq[j]))
408                .max_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal))
409                .expect("n_list > 0");
410            assignments[i] = best_j as u32;
411        }
412        Ok(assignments)
413    }

415    fn train_kmeans(
416        &self,
417        flat_sample: &[f32],
418        n_samples: usize,
419        n_iter: usize,
420    ) -> Result<Vec<f32>> {
421        let mut centroids = kmeans_plus_plus_init(
422            flat_sample,
423            n_samples,
424            self.dimension,
425            self.config.n_list,
426            self.config.seed,
427        );
428        let mut prev_inertia = f64::INFINITY;
429        for iter in 0..n_iter {
430            let assignments = self.assign_to_centroids(flat_sample, n_samples, &centroids)?;
431            let (new_centroids, inertia) = update_centroids(
432                flat_sample,
433                n_samples,
434                &assignments,
435                &centroids,
436                self.dimension,
437                self.config.n_list,
438            );
439            centroids = new_centroids;
440            debug!("kmeans iter {iter}: inertia={inertia:.6}");
441            if (prev_inertia - inertia).abs() <= 1e-6 * prev_inertia.abs().max(1.0) {
442                debug!("kmeans converged after {} iters", iter + 1);
443                break;
444            }
445            prev_inertia = inertia;
446        }
447        Ok(centroids)
448    }

450    fn score_from_dot(&self, dot: f32, local_idx: usize, query_norm_sq: f32) -> f32 {
451        match self.metric {
452            GpuDistanceMetric::DotProduct => dot,
453            GpuDistanceMetric::Cosine => {
454                let v_norm = self.vector_norms_sq[local_idx].sqrt();
455                let q_norm = query_norm_sq.sqrt();
456                let denom = v_norm * q_norm;
457                if denom > 0.0 { dot / denom } else { 0.0 }
458            }
459            GpuDistanceMetric::Euclidean => {
460                (self.vector_norms_sq[local_idx] - 2.0 * dot + query_norm_sq).max(0.0)
461            }
462        }
463    }
464
465    fn finalize_top_k(&self, mut scored: Vec<(usize, f32)>, k: usize) -> Vec<GpuSearchResult> {
466        match self.metric {
467            GpuDistanceMetric::Euclidean => {
468                scored.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal))
469            }
470            _ => scored.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal)),
471        }
472        scored.truncate(k);
473        scored
474            .into_iter()
475            .map(|(index, score)| GpuSearchResult {
476                id: self.ids_by_local_index[index].clone(),
477                score: match self.metric {
478                    GpuDistanceMetric::Euclidean => 1.0 / (1.0 + score.sqrt()),
479                    _ => score,
480                },
481                index,
482            })
483            .collect()
484    }

498// ---- shared helpers (copied from CUDA/ROCm IVFs; kept local for now so
499//       the three backends can evolve independently) ---------------------

501#[inline]
502fn dot_self(v: &[f32]) -> f32 {
503    v.iter().map(|&x| x * x).sum()
504}
505
506#[inline]
507fn l2_sq(a: &[f32], b: &[f32]) -> f32 {
508    a.iter().zip(b).map(|(x, y)| (x - y).powi(2)).sum()
509}
510
511fn select_nprobe_clusters(
512    dots: &[f32],
513    centroid_norms_sq: &[f32],
514    _query_norm_sq: f32,
515    nprobe: usize,
516) -> Vec<usize> {
517    let mut scored: Vec<(usize, f32)> = dots
518        .iter()
519        .enumerate()
520        .map(|(i, &dot)| (i, centroid_norms_sq[i] - 2.0 * dot))
521        .collect();
522    scored.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal));
523    scored.truncate(nprobe);
524    scored.into_iter().map(|(i, _)| i).collect()
525}

527fn build_cluster_layout(assignments: &[u32], n_list: usize) -> (Vec<usize>, Vec<usize>) {
528    let mut counts = vec![0usize; n_list];
529    for &a in assignments {
530        counts[a as usize] += 1;
531    }
532    let mut offsets = Vec::with_capacity(n_list + 1);
533    offsets.push(0);
534    for c in &counts {
535        offsets.push(*offsets.last().unwrap() + c);
536    }
537    let mut perm = vec![0usize; assignments.len()];
538    let mut cursors = offsets.clone();
539    for (global_idx, &a) in assignments.iter().enumerate() {
540        let pos = cursors[a as usize];
541        perm[pos] = global_idx;
542        cursors[a as usize] += 1;
543    }
544    (offsets, perm)
545}

547fn kmeans_plus_plus_init(
548    flat_sample: &[f32],
549    n_samples: usize,
550    dimension: usize,
551    n_list: usize,
552    seed: Option<u64>,
553) -> Vec<f32> {
554    let mut rng = SplitMix64::new(seed.unwrap_or(0x9E37_79B9_7F4A_7C15));
555    let mut centroids = Vec::with_capacity(n_list * dimension);
556    let first = (rng.next_u64() as usize) % n_samples;
557    centroids.extend_from_slice(&flat_sample[first * dimension..(first + 1) * dimension]);
558    let mut min_dist_sq = vec![f32::INFINITY; n_samples];
559    for c in 0..n_list - 1 {
560        let last = &centroids[c * dimension..(c + 1) * dimension];
561        for i in 0..n_samples {
562            let d = l2_sq(&flat_sample[i * dimension..(i + 1) * dimension], last);
563            if d < min_dist_sq[i] {
564                min_dist_sq[i] = d;
565            }
566        }
567        let total: f64 = min_dist_sq.iter().map(|&x| x as f64).sum();
568        if total <= 0.0 {
569            let pick = (rng.next_u64() as usize) % n_samples;
570            centroids.extend_from_slice(&flat_sample[pick * dimension..(pick + 1) * dimension]);
571            continue;
572        }
573        let target = (rng.next_f64() * total) as f32;
574        let mut acc = 0f32;
575        let mut pick = n_samples - 1;
576        for (i, &d) in min_dist_sq.iter().enumerate() {
577            acc += d;
578            if acc >= target {
579                pick = i;
580                break;
581            }
582        }
583        centroids.extend_from_slice(&flat_sample[pick * dimension..(pick + 1) * dimension]);
584    }
585    centroids
586}

588fn update_centroids(
589    flat_sample: &[f32],
590    n_samples: usize,
591    assignments: &[u32],
592    centroids: &[f32],
593    dimension: usize,
594    n_list: usize,
595) -> (Vec<f32>, f64) {
596    let mut sums = vec![0f32; n_list * dimension];
597    let mut counts = vec![0usize; n_list];
598    for (i, &assigned) in assignments.iter().enumerate().take(n_samples) {
599        let c = assigned as usize;
600        counts[c] += 1;
601        let base = c * dimension;
602        let sbase = i * dimension;
603        for d in 0..dimension {
604            sums[base + d] += flat_sample[sbase + d];
605        }
606    }
607    let mut new_centroids = centroids.to_vec();
608    for j in 0..n_list {
609        if counts[j] == 0 {
610            let mut worst = 0usize;
611            let mut worst_d = -1f32;
612            for i in 0..n_samples {
613                let a = assignments[i] as usize;
614                let c = &centroids[a * dimension..(a + 1) * dimension];
615                let d = l2_sq(&flat_sample[i * dimension..(i + 1) * dimension], c);
616                if d > worst_d {
617                    worst_d = d;
618                    worst = i;
619                }
620            }
621            new_centroids[j * dimension..(j + 1) * dimension]
622                .copy_from_slice(&flat_sample[worst * dimension..(worst + 1) * dimension]);
623            continue;
624        }
625        let inv = 1.0 / counts[j] as f32;
626        for d in 0..dimension {
627            new_centroids[j * dimension + d] = sums[j * dimension + d] * inv;
628        }
629    }
630    let mut inertia = 0f64;
631    for i in 0..n_samples {
632        let j = assignments[i] as usize;
633        let d = l2_sq(
634            &flat_sample[i * dimension..(i + 1) * dimension],
635            &new_centroids[j * dimension..(j + 1) * dimension],
636        );
637        inertia += d as f64;
638    }
639    (new_centroids, inertia)
640}

642#[derive(Debug, Clone, Copy)]
643struct SplitMix64 {
644    state: u64,
645}
646impl SplitMix64 {
647    fn new(seed: u64) -> Self {
648        Self { state: seed }
649    }
650    fn next_u64(&mut self) -> u64 {
651        self.state = self.state.wrapping_add(0x9E37_79B9_7F4A_7C15);
652        let mut z = self.state;
653        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
654        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
655        z ^ (z >> 31)
656    }
657    fn next_f64(&mut self) -> f64 {
658        (self.next_u64() >> 11) as f64 / ((1u64 << 53) as f64)
659    }
