1//! Intel GPU backend via Vulkan Compute.
2//!
3//! Gated behind the `intel` Cargo feature and compiled on Linux or
4//! Windows. The Vulkan loader is dlopened at runtime by `ash`, so the
5//! crate builds on any host — the feature only becomes functional when a
6//! Vulkan driver is present.
7//!
8//! ⚠️ AUTHORED BLIND — no Intel Arc / Battlemage hardware was available
9//! at write time. See `phase3c_add-intel-backend`.
10
11#![cfg(all(feature = "intel", any(target_os = "linux", target_os = "windows")))]
12// Vulkan dispatch boilerplate benefits from pedantic-style rewrites but
13// rewriting every `(x + 255) / 256` to `.div_ceil(256)` and every
14// `slice.len() * size_of::<T>()` to `size_of_val` obscures the intent
15// for a backend that is still evolving. Re-enable these once the Intel
16// maintainer validates the code on real hardware.
17#![allow(clippy::manual_div_ceil)]
18#![allow(clippy::manual_slice_size_calculation)]
19#![allow(clippy::manual_is_multiple_of)]
20
21pub mod context;
22pub mod ivf;
23pub mod vector_storage;
24
25pub use context::IntelContext;
26pub use ivf::IntelIvfIndex;
27pub use vector_storage::IntelVectorStorage;
