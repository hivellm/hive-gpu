1//! # Intel (Vulkan) Vector Storage
2//!
3//! Stores vector payloads in a `HOST_VISIBLE | HOST_COHERENT` Vulkan
4//! buffer. Brute-force search dispatches the `sgemv_dot` compute pipeline
5//! built during context creation; metric post-processing and top-K run on
6//! the host. Mirrors the shape of the CUDA, ROCm, and Metal backends.
7//!
8//! Memory strategy: for simplicity and because Intel GPUs (both
9//! integrated Xe and Arc) handle coherent memory well, every buffer is
10//! allocated as `HOST_VISIBLE | HOST_COHERENT` rather than staging
11//! through device-local memory. A future optimisation can introduce a
12//! staging + blit pattern matching the CUDA and ROCm backends if
13//! benchmarks show it matters.
14//!
15//! ⚠️ AUTHORED BLIND — see phase3c_add-intel-backend.
16

31/// Vulkan buffer + backing device memory, kept together so drop order
32/// stays correct.
33pub(crate) struct VulkanBuffer {
34    pub buffer: vk::Buffer,
35    pub memory: vk::DeviceMemory,
36    pub size_bytes: usize,
37    /// Cached host-mapped pointer for `HOST_VISIBLE` buffers. Null when
38    /// the buffer is device-local only.
39    pub host_ptr: *mut u8,
40}
41
42// SAFETY: the raw pointer is only used through the device (Vulkan
43// serialises it) and we never share it between threads without external
44// synchronisation.
45unsafe impl Send for VulkanBuffer {}
46unsafe impl Sync for VulkanBuffer {}
47
48impl VulkanBuffer {
49    pub(crate) fn destroy(self, device: &ash::Device) {
50        // SAFETY: both objects were created by us and no other thread is
51        // accessing them (the caller holds exclusive ownership).
52        unsafe {
53            if !self.host_ptr.is_null() {
54                device.unmap_memory(self.memory);
55            }
56            device.destroy_buffer(self.buffer, None);
57            device.free_memory(self.memory, None);
58        }
59    }
60}

62/// Allocate a host-visible, host-coherent Vulkan buffer of at least
63/// `size_bytes` bytes usable as a compute-shader storage buffer. Mapped
64/// pointer is kept alive for the buffer's lifetime so `write_f32_slice`
65/// and `read_f32_slice` can copy without remapping.
66pub(crate) fn allocate_host_visible_buffer(
67    context: &IntelContext,
68    size_bytes: usize,
69) -> Result<VulkanBuffer> {
70    let size_bytes = size_bytes.max(1);
71    let device = context.device();
72
73    let buffer_info = vk::BufferCreateInfo::default()
74        .size(size_bytes as u64)
75        .usage(
76            vk::BufferUsageFlags::STORAGE_BUFFER
77                | vk::BufferUsageFlags::TRANSFER_SRC
78                | vk::BufferUsageFlags::TRANSFER_DST,
79        )
80        .sharing_mode(vk::SharingMode::EXCLUSIVE);
81
82    // SAFETY: `buffer_info` scoped; `device` outlives the returned buffer
83    // as long as the VulkanBuffer handle stays alive.
84    let buffer = unsafe { device.create_buffer(&buffer_info, None) }
85        .map_err(|e| HiveGpuError::VulkanError(format!("create_buffer: {e:?}")))?;
86
87    // SAFETY: buffer just created.
88    let reqs = unsafe { device.get_buffer_memory_requirements(buffer) };
89    let memory_type_index = pick_memory_type(
90        context,
91        reqs.memory_type_bits,
92        vk::MemoryPropertyFlags::HOST_VISIBLE | vk::MemoryPropertyFlags::HOST_COHERENT,
93    )
94    .ok_or_else(|| {
95        HiveGpuError::VulkanError("no host-visible memory type available".to_string())
96    })?;
97
98    let alloc_info = vk::MemoryAllocateInfo::default()
99        .allocation_size(reqs.size)
100        .memory_type_index(memory_type_index);
101    // SAFETY: alloc_info scoped.
102    let memory = unsafe { device.allocate_memory(&alloc_info, None) }.map_err(|e| {
103        // SAFETY: buffer was created above; cleanup on failure.
104        unsafe {
105            device.destroy_buffer(buffer, None);
106        }
107        HiveGpuError::VulkanError(format!("allocate_memory: {e:?}"))
108    })?;
109
110    // SAFETY: buffer + memory just created; memory type selected to be
111    // compatible with buffer.
112    unsafe {
113        device
114            .bind_buffer_memory(buffer, memory, 0)
115            .map_err(|e| HiveGpuError::VulkanError(format!("bind_buffer_memory: {e:?}")))?;
116    }
117
118    // Map for host access.
119    // SAFETY: HOST_VISIBLE type selected; mapping the entire range is
120    // supported for coherent memory.
121    let host_ptr = unsafe {
122        device
123            .map_memory(memory, 0, reqs.size, vk::MemoryMapFlags::empty())
124            .map_err(|e| HiveGpuError::VulkanError(format!("map_memory: {e:?}")))?
125            as *mut u8
126    };
127
128    Ok(VulkanBuffer {
129        buffer,
130        memory,
131        size_bytes: reqs.size as usize,
132        host_ptr,
133    })
134}

136fn pick_memory_type(
137    context: &IntelContext,
138    type_bits: u32,
139    required: vk::MemoryPropertyFlags,
140) -> Option<u32> {
141    let props = context.memory_properties();
142    (0..props.memory_type_count).find(|i| {
143        let ty = props.memory_types[*i as usize];
144        (type_bits & (1 << i)) != 0 && ty.property_flags.contains(required)
145    })
146}

148/// Write an f32 slice into the mapped region of a host-visible buffer
149/// (overwrites from offset 0).
150pub(crate) fn write_f32_slice(buf: &VulkanBuffer, data: &[f32]) -> Result<()> {
151    if buf.host_ptr.is_null() {
152        return Err(HiveGpuError::VulkanError(
153            "write_f32_slice requires a host-visible buffer".to_string(),
154        ));
155    }
156    let bytes = data.len() * std::mem::size_of::<f32>();
157    if bytes > buf.size_bytes {
158        return Err(HiveGpuError::VulkanError(format!(
159            "write overflows buffer: {bytes} > {size}",
160            size = buf.size_bytes
161        )));
162    }
163    // SAFETY: `host_ptr` is mapped for at least `buf.size_bytes`; we bound
164    // the write by `bytes`; HOST_COHERENT memory means no explicit flush
165    // is required before the next queue submit reads the data.
166    unsafe {
167        std::ptr::copy_nonoverlapping(data.as_ptr() as *const u8, buf.host_ptr, bytes);
168    }
169    Ok(())
170}

172/// Write a slice starting at a given byte offset inside the mapped region.
173pub(crate) fn write_f32_slice_at(
174    buf: &VulkanBuffer,
175    byte_offset: usize,
176    data: &[f32],
177) -> Result<()> {
178    if buf.host_ptr.is_null() {
179        return Err(HiveGpuError::VulkanError(
180            "write_f32_slice_at requires a host-visible buffer".to_string(),
181        ));
182    }
183    let bytes = data.len() * std::mem::size_of::<f32>();
184    if byte_offset + bytes > buf.size_bytes {
185        return Err(HiveGpuError::VulkanError(format!(
186            "offset write overflows buffer: {byte_offset}+{bytes} > {size}",
187            size = buf.size_bytes
188        )));
189    }
190    // SAFETY: bounds checked above.
191    unsafe {
192        std::ptr::copy_nonoverlapping(
193            data.as_ptr() as *const u8,
194            buf.host_ptr.add(byte_offset),
195            bytes,
196        );
197    }
198    Ok(())
199}

201/// Read a contiguous prefix of a host-visible buffer into a Rust `Vec<f32>`.
202pub(crate) fn read_f32_vec(buf: &VulkanBuffer, count: usize) -> Result<Vec<f32>> {
203    if buf.host_ptr.is_null() {
204        return Err(HiveGpuError::VulkanError(
205            "read_f32_vec requires a host-visible buffer".to_string(),
206        ));
207    }
208    let bytes = count * std::mem::size_of::<f32>();
209    if bytes > buf.size_bytes {
210        return Err(HiveGpuError::VulkanError(format!(
211            "read overflows buffer: {bytes} > {size}",
212            size = buf.size_bytes
213        )));
214    }
215    let mut out = vec![0f32; count];
216    // SAFETY: bounds checked.
217    unsafe {
218        std::ptr::copy_nonoverlapping(
219            buf.host_ptr as *const u8,
220            out.as_mut_ptr() as *mut u8,
221            bytes,
222        );
223    }
224    Ok(out)
225}

227/// Device-to-device `vkCmdCopyBuffer` dispatched on a one-shot command
228/// buffer. Used by storage growth and by anything that needs to move
229/// bytes between two host-visible buffers without touching the host.
230pub(crate) fn dtod_copy(
231    context: &IntelContext,
232    src: &VulkanBuffer,
233    dst: &VulkanBuffer,
234    bytes: usize,
235) -> Result<()> {
236    if bytes == 0 {
237        return Ok(());
238    }
239    let device = context.device();
240    let cb = begin_one_shot_command(context)?;
241    let region = [vk::BufferCopy {
242        src_offset: 0,
243        dst_offset: 0,
244        size: bytes as u64,
245    }];
246    // SAFETY: command buffer recorded inside the scope we just opened.
247    unsafe {
248        device.cmd_copy_buffer(cb, src.buffer, dst.buffer, &region);
249    }
250    end_and_submit_one_shot(context, cb)
251}

253/// Dispatch a compute pipeline over three storage buffers with a push
254/// constants payload and a grid size. Allocates a one-shot descriptor
255/// set + command buffer + fence; every operation waits to completion
256/// before returning. Simple and slow; optimisation is a follow-up.
257pub(crate) fn dispatch_three_buffer_compute<P: bytemuck_like::PodLike>(
258    context: &IntelContext,
259    pipeline: ComputePipeline,
260    buffers: [&VulkanBuffer; 3],
261    push_constants: P,
262    grid: (u32, u32, u32),
263) -> Result<()> {
264    let device = context.device();
265
266    // 1. Allocate a descriptor set from the context's pool.
267    let layouts = [pipeline.set_layout];
268    let ds_alloc = vk::DescriptorSetAllocateInfo::default()
269        .descriptor_pool(context.descriptor_pool())
270        .set_layouts(&layouts);
271    // SAFETY: layouts slice scoped; descriptor pool has FREE flag set.
272    let descriptor_sets = unsafe { device.allocate_descriptor_sets(&ds_alloc) }
273        .map_err(|e| HiveGpuError::VulkanError(format!("allocate_descriptor_sets: {e:?}")))?;
274    let descriptor_set = descriptor_sets[0];
275
276    // 2. Bind the three storage buffers.
277    let buffer_infos: [vk::DescriptorBufferInfo; 3] = [
278        vk::DescriptorBufferInfo::default()
279            .buffer(buffers[0].buffer)
280            .offset(0)
281            .range(vk::WHOLE_SIZE),
282        vk::DescriptorBufferInfo::default()
283            .buffer(buffers[1].buffer)
284            .offset(0)
285            .range(vk::WHOLE_SIZE),
286        vk::DescriptorBufferInfo::default()
287            .buffer(buffers[2].buffer)
288            .offset(0)
289            .range(vk::WHOLE_SIZE),
290    ];
291    let writes = [
292        vk::WriteDescriptorSet::default()
293            .dst_set(descriptor_set)
294            .dst_binding(0)
295            .descriptor_type(vk::DescriptorType::STORAGE_BUFFER)
296            .buffer_info(std::slice::from_ref(&buffer_infos[0])),
297        vk::WriteDescriptorSet::default()
298            .dst_set(descriptor_set)
299            .dst_binding(1)
300            .descriptor_type(vk::DescriptorType::STORAGE_BUFFER)
301            .buffer_info(std::slice::from_ref(&buffer_infos[1])),
302        vk::WriteDescriptorSet::default()
303            .dst_set(descriptor_set)
304            .dst_binding(2)
305            .descriptor_type(vk::DescriptorType::STORAGE_BUFFER)
306            .buffer_info(std::slice::from_ref(&buffer_infos[2])),
307    ];
308    // SAFETY: all slices scoped.
309    unsafe {
310        device.update_descriptor_sets(&writes, &[]);
311    }
312
313    // 3. Record + submit + wait.
314    let cb = begin_one_shot_command(context)?;
315    // SAFETY: recording a one-shot command buffer we just allocated.
316    unsafe {
317        device.cmd_bind_pipeline(cb, vk::PipelineBindPoint::COMPUTE, pipeline.pipeline);
318        device.cmd_bind_descriptor_sets(
319            cb,
320            vk::PipelineBindPoint::COMPUTE,
321            pipeline.layout,
322            0,
323            &[descriptor_set],
324            &[],
325        );
326        let pc_bytes = std::slice::from_raw_parts(
327            &push_constants as *const P as *const u8,
328            std::mem::size_of::<P>(),
329        );
330        device.cmd_push_constants(
331            cb,
332            pipeline.layout,
333            vk::ShaderStageFlags::COMPUTE,
334            0,
335            pc_bytes,
336        );
337        device.cmd_dispatch(cb, grid.0, grid.1, grid.2);
338    }
339    end_and_submit_one_shot(context, cb)?;
340
341    // 4. Free the descriptor set (pool was created with FREE_DESCRIPTOR_SET).
342    // SAFETY: descriptor set allocated from this exact pool.
343    unsafe {
344        device
345            .free_descriptor_sets(context.descriptor_pool(), &[descriptor_set])
346            .map_err(|e| HiveGpuError::VulkanError(format!("free_descriptor_sets: {e:?}")))?;
347    }
348
349    Ok(())
350}

352fn begin_one_shot_command(context: &IntelContext) -> Result<vk::CommandBuffer> {
353    let device = context.device();
354    let alloc = vk::CommandBufferAllocateInfo::default()
355        .command_pool(context.command_pool())
356        .level(vk::CommandBufferLevel::PRIMARY)
357        .command_buffer_count(1);
358    // SAFETY: alloc scoped.
359    let buffers = unsafe { device.allocate_command_buffers(&alloc) }
360        .map_err(|e| HiveGpuError::VulkanError(format!("allocate_command_buffers: {e:?}")))?;
361    let cb = buffers[0];
362    let begin =
363        vk::CommandBufferBeginInfo::default().flags(vk::CommandBufferUsageFlags::ONE_TIME_SUBMIT);
364    // SAFETY: cb just allocated.
365    unsafe { device.begin_command_buffer(cb, &begin) }
366        .map_err(|e| HiveGpuError::VulkanError(format!("begin_command_buffer: {e:?}")))?;
367    Ok(cb)
368}
369
370fn end_and_submit_one_shot(context: &IntelContext, cb: vk::CommandBuffer) -> Result<()> {
371    let device = context.device();
372    // SAFETY: cb is mid-recording.
373    unsafe {
374        device
375            .end_command_buffer(cb)
376            .map_err(|e| HiveGpuError::VulkanError(format!("end_command_buffer: {e:?}")))?;
377    }
378
379    // Create a fence to wait on.
380    let fence_info = vk::FenceCreateInfo::default();
381    // SAFETY: fence_info scoped.
382    let fence = unsafe { device.create_fence(&fence_info, None) }
383        .map_err(|e| HiveGpuError::VulkanError(format!("create_fence: {e:?}")))?;
384
385    let cmd_buffers = [cb];
386    let submit = [vk::SubmitInfo::default().command_buffers(&cmd_buffers)];
387    // SAFETY: submit + cmd_buffers scoped; fence just created.
388    unsafe {
389        device
390            .queue_submit(context.queue(), &submit, fence)
391            .map_err(|e| HiveGpuError::VulkanError(format!("queue_submit: {e:?}")))?;
392        device
393            .wait_for_fences(&[fence], true, u64::MAX)
394            .map_err(|e| HiveGpuError::VulkanError(format!("wait_for_fences: {e:?}")))?;
395        device.destroy_fence(fence, None);
396        device.free_command_buffers(context.command_pool(), &cmd_buffers);
397    }
398    Ok(())
399}

401/// Same as [`dispatch_three_buffer_compute`] but with an explicit byte
402/// offset + range per binding, so callers can expose a sub-range of a
403/// larger buffer. Used by the IVF index for per-cluster refined search.
404pub(crate) fn dispatch_three_buffer_compute_ranged<P: bytemuck_like::PodLike>(
405    context: &IntelContext,
406    pipeline: ComputePipeline,
407    bindings: [(&VulkanBuffer, u64, u64); 3],
408    push_constants: P,
409    grid: (u32, u32, u32),
410) -> Result<()> {
411    let device = context.device();
412
413    let layouts = [pipeline.set_layout];
414    let ds_alloc = vk::DescriptorSetAllocateInfo::default()
415        .descriptor_pool(context.descriptor_pool())
416        .set_layouts(&layouts);
417    let descriptor_sets = unsafe { device.allocate_descriptor_sets(&ds_alloc) }
418        .map_err(|e| HiveGpuError::VulkanError(format!("allocate_descriptor_sets: {e:?}")))?;
419    let descriptor_set = descriptor_sets[0];
420
421    let buffer_infos: [vk::DescriptorBufferInfo; 3] = [
422        vk::DescriptorBufferInfo::default()
423            .buffer(bindings[0].0.buffer)
424            .offset(bindings[0].1)
425            .range(bindings[0].2),
426        vk::DescriptorBufferInfo::default()
427            .buffer(bindings[1].0.buffer)
428            .offset(bindings[1].1)
429            .range(bindings[1].2),
430        vk::DescriptorBufferInfo::default()
431            .buffer(bindings[2].0.buffer)
432            .offset(bindings[2].1)
433            .range(bindings[2].2),
434    ];
435    let writes = [
436        vk::WriteDescriptorSet::default()
437            .dst_set(descriptor_set)
438            .dst_binding(0)
439            .descriptor_type(vk::DescriptorType::STORAGE_BUFFER)
440            .buffer_info(std::slice::from_ref(&buffer_infos[0])),
441        vk::WriteDescriptorSet::default()
442            .dst_set(descriptor_set)
443            .dst_binding(1)
444            .descriptor_type(vk::DescriptorType::STORAGE_BUFFER)
445            .buffer_info(std::slice::from_ref(&buffer_infos[1])),
446        vk::WriteDescriptorSet::default()
447            .dst_set(descriptor_set)
448            .dst_binding(2)
449            .descriptor_type(vk::DescriptorType::STORAGE_BUFFER)
450            .buffer_info(std::slice::from_ref(&buffer_infos[2])),
451    ];
452    unsafe {
453        device.update_descriptor_sets(&writes, &[]);
454    }
455
456    let cb = begin_one_shot_command(context)?;
457    unsafe {
458        device.cmd_bind_pipeline(cb, vk::PipelineBindPoint::COMPUTE, pipeline.pipeline);
459        device.cmd_bind_descriptor_sets(
460            cb,
461            vk::PipelineBindPoint::COMPUTE,
462            pipeline.layout,
463            0,
464            &[descriptor_set],
465            &[],
466        );
467        let pc_bytes = std::slice::from_raw_parts(
468            &push_constants as *const P as *const u8,
469            std::mem::size_of::<P>(),
470        );
471        device.cmd_push_constants(
472            cb,
473            pipeline.layout,
474            vk::ShaderStageFlags::COMPUTE,
475            0,
476            pc_bytes,
477        );
478        device.cmd_dispatch(cb, grid.0, grid.1, grid.2);
479    }
480    end_and_submit_one_shot(context, cb)?;
481
482    unsafe {
483        device
484            .free_descriptor_sets(context.descriptor_pool(), &[descriptor_set])
485            .map_err(|e| HiveGpuError::VulkanError(format!("free_descriptor_sets: {e:?}")))?;
486    }
487
488    Ok(())
489}

519impl std::fmt::Debug for IntelVectorStorage {

521        f.debug_struct("IntelVectorStorage")
522            .field("vector_count", &self.vector_count)
523            .field("buffer_capacity", &self.buffer_capacity)

525            .field("metric", &self.metric)
526            .field("removed", &self.removed_indices.len())
527            .finish()

530
531impl IntelVectorStorage {
532    pub fn new(
533        context: Arc<IntelContext>,
534        dimension: usize,
535        metric: GpuDistanceMetric,
536    ) -> Result<Self> {
537        if dimension == 0 {
538            return Err(HiveGpuError::InvalidConfiguration(
539                "dimension must be > 0".to_string(),
540            ));
541        }
542
543        let min_vectors_by_bytes =
544            (MIN_INITIAL_BYTES / (dimension * std::mem::size_of::<f32>())).max(1);
545        let capacity = MIN_INITIAL_VECTORS.max(min_vectors_by_bytes);
546        let slots = capacity
547            .checked_mul(dimension)
548            .ok_or_else(|| HiveGpuError::InvalidConfiguration("capacity overflow".to_string()))?;
549        let bytes = slots * std::mem::size_of::<f32>();
550
551        let storage = allocate_host_visible_buffer(&context, bytes)?;
552
553        debug!(
554            "intel storage created: dim={} capacity={} bytes={}",
555            dimension, capacity, bytes
556        );
557
558        Ok(Self {
559            context,
560            storage: Some(storage),
561            buffer_capacity: capacity,
562            vector_count: 0,
563            dimension,
564            metric,
565            vector_id_map: HashMap::new(),
566            index_to_id: Vec::new(),
567            removed_indices: HashSet::new(),
568            payloads: HashMap::new(),
569            norms_sq: Vec::new(),
570        })
571    }
572
573    fn validate_vector(&self, vector: &GpuVector) -> Result<()> {
574        if vector.data.len() != self.dimension {
575            return Err(HiveGpuError::DimensionMismatch {
576                expected: self.dimension,
577                actual: vector.data.len(),
578            });
579        }
580        if vector.id.is_empty() {
581            return Err(HiveGpuError::InvalidConfiguration(
582                "vector id must be non-empty".to_string(),
583            ));
584        }
585        if vector.id.len() > 256 {
586            return Err(HiveGpuError::InvalidConfiguration(
587                "vector id must be <= 256 chars".to_string(),
588            ));
589        }
590        if self.vector_id_map.contains_key(&vector.id) {
591            return Err(HiveGpuError::InvalidConfiguration(format!(
592                "duplicate vector id: {}",
593                vector.id
594            )));
595        }
596        for (i, &v) in vector.data.iter().enumerate() {
597            if !v.is_finite() {
598                return Err(HiveGpuError::InvalidConfiguration(format!(
599                    "non-finite component at index {i} in vector {}",
600                    vector.id
601                )));
602            }
603        }
604        Ok(())
605    }
606
607    fn ensure_capacity(&mut self, additional: usize) -> Result<()> {
608        let required = self
609            .vector_count
610            .checked_add(additional)
611            .ok_or_else(|| HiveGpuError::InvalidConfiguration("capacity overflow".to_string()))?;
612        if required <= self.buffer_capacity {
613            return Ok(());
614        }
615        let mut new_capacity = self.buffer_capacity;
616        while new_capacity < required {
617            let factor = if new_capacity < 1_000 {
618                2.0f32
619            } else if new_capacity < 10_000 {
620                1.5f32
621            } else {
622                1.2f32
623            };
624            new_capacity = ((new_capacity as f32) * factor).ceil() as usize;
625            new_capacity = new_capacity.max(required);
626        }
627        let slots = new_capacity
628            .checked_mul(self.dimension)
629            .ok_or_else(|| HiveGpuError::InvalidConfiguration("slots overflow".to_string()))?;
630        let bytes = slots * std::mem::size_of::<f32>();
631
632        let new_buf = allocate_host_visible_buffer(&self.context, bytes)?;
633        if self.vector_count > 0 {
634            let live_bytes = self.vector_count * self.dimension * std::mem::size_of::<f32>();
635            let old = self.storage.as_ref().expect("storage live");
636            dtod_copy(&self.context, old, &new_buf, live_bytes)?;
637        }
638        if let Some(old) = self.storage.take() {
639            old.destroy(self.context.device());
640        }
641        info!(
642            "intel storage expand: {} -> {} vectors ({:.2} MiB)",
643            self.buffer_capacity,
644            new_capacity,
645            bytes as f64 / (1024.0 * 1024.0)
646        );
647        self.storage = Some(new_buf);
648        self.buffer_capacity = new_capacity;
649        Ok(())
650    }

664    /// Run the `sgemv_dot` pipeline against the full stored vector
665    /// buffer. Returns one dot product per stored vector.
666    pub(crate) fn gpu_scores(&self, query: &[f32]) -> Result<Vec<f32>> {
667        if self.vector_count == 0 {
668            return Ok(Vec::new());
669        }
670        if query.len() != self.dimension {
671            return Err(HiveGpuError::DimensionMismatch {
672                expected: self.dimension,
673                actual: query.len(),
674            });
675        }
676        for (i, &v) in query.iter().enumerate() {
677            if !v.is_finite() {
678                return Err(HiveGpuError::InvalidConfiguration(format!(
679                    "non-finite query component at index {i}"
680                )));
681            }
682        }
683
684        let query_bytes = query.len() * std::mem::size_of::<f32>();
685        let query_buf = allocate_host_visible_buffer(&self.context, query_bytes)?;
686        write_f32_slice(&query_buf, query)?;
687        let scores_bytes = self.vector_count * std::mem::size_of::<f32>();
688        let scores_buf = allocate_host_visible_buffer(&self.context, scores_bytes)?;
689
690        let pipeline = self.context.sgemv_dot();
691        let pc = SgemvPushConstants {
692            dimension: self.dimension as u32,
693            n_vectors: self.vector_count as u32,
694        };
695        let grid = (((self.vector_count as u32) + 255) / 256, 1, 1);
696        let storage_buf = self.storage.as_ref().expect("storage live");
697        dispatch_three_buffer_compute(
698            &self.context,
699            pipeline,
700            [storage_buf, &query_buf, &scores_buf],
701            pc,
702            grid,
703        )?;
704
705        let scores = read_f32_vec(&scores_buf, self.vector_count)?;
706
707        query_buf.destroy(self.context.device());
708        scores_buf.destroy(self.context.device());
709        Ok(scores)
710    }
711
712    fn apply_metric(&self, raw_scores: &mut [f32], query: &[f32]) {
713        let query_norm_sq: f32 = query.iter().map(|&x| x * x).sum();
714        match self.metric {
715            GpuDistanceMetric::DotProduct => {}
716            GpuDistanceMetric::Cosine => {
717                let q_norm = query_norm_sq.sqrt();
718                for (i, s) in raw_scores.iter_mut().enumerate() {
719                    let v_norm = self.norms_sq[i].sqrt();
720                    let denom = v_norm * q_norm;
721                    *s = if denom > 0.0 { *s / denom } else { 0.0 };
722                }
723            }
724            GpuDistanceMetric::Euclidean => {
725                for (i, s) in raw_scores.iter_mut().enumerate() {
726                    *s = (self.norms_sq[i] - 2.0 * *s + query_norm_sq).max(0.0);
727                }
728            }
729        }
730    }
731
732    fn select_top_k(&self, mut scored: Vec<(usize, f32)>, limit: usize) -> Vec<GpuSearchResult> {
733        scored.retain(|(idx, _)| !self.removed_indices.contains(idx));
734        match self.metric {
735            GpuDistanceMetric::Euclidean => {
736                scored.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal))
737            }
738            _ => scored.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal)),
739        }
740        scored.truncate(limit);
741        scored
742            .into_iter()
743            .map(|(index, score)| {
744                let id = self.index_to_id[index].clone();
745                let similarity = match self.metric {
746                    GpuDistanceMetric::Euclidean => 1.0 / (1.0 + score.sqrt()),
747                    _ => score,
748                };
749                GpuSearchResult {
750                    id,
751                    score: similarity,
752                    index,
753                }
754            })
755            .collect()
756    }

767impl GpuVectorStorage for IntelVectorStorage {
768    fn add_vectors(&mut self, vectors: &[GpuVector]) -> Result<Vec<usize>> {
769        if vectors.is_empty() {
770            return Ok(Vec::new());
771        }
772        let mut seen = HashSet::with_capacity(vectors.len());
773        for v in vectors {
774            self.validate_vector(v)?;
775            if !seen.insert(v.id.as_str()) {
776                return Err(HiveGpuError::InvalidConfiguration(format!(
777                    "duplicate vector id within batch: {}",
778                    v.id
779                )));
780            }
781        }
782        self.ensure_capacity(vectors.len())?;
783
784        let mut flat = Vec::with_capacity(vectors.len() * self.dimension);
785        for v in vectors {
786            flat.extend_from_slice(&v.data);
787        }
788        let offset_bytes = self.vector_count * self.dimension * std::mem::size_of::<f32>();
789        let storage = self.storage.as_ref().expect("storage live");
790        write_f32_slice_at(storage, offset_bytes, &flat)?;
791
792        let mut indices = Vec::with_capacity(vectors.len());
793        for v in vectors {
794            let index = self.vector_count;
795            self.vector_id_map.insert(v.id.clone(), index);
796            self.index_to_id.push(v.id.clone());
797            self.payloads.insert(v.id.clone(), v.metadata.clone());
798            self.norms_sq
799                .push(v.data.iter().map(|&x| x * x).sum::<f32>());
800            self.vector_count += 1;
801            indices.push(index);
802        }
803        Ok(indices)
804    }
805
806    fn search(&self, query: &[f32], limit: usize) -> Result<Vec<GpuSearchResult>> {
807        if limit == 0 || self.vector_count == 0 {
808            return Ok(Vec::new());
809        }
810        let mut scores = self.gpu_scores(query)?;
811        self.apply_metric(&mut scores, query);
812        let scored: Vec<(usize, f32)> = scores.into_iter().enumerate().collect();
813        Ok(self.select_top_k(scored, limit))
814    }
815
816    fn remove_vectors(&mut self, ids: &[String]) -> Result<()> {
817        for id in ids {
818            if let Some(&index) = self.vector_id_map.get(id) {
819                self.removed_indices.insert(index);
820                self.payloads.remove(id);
821            } else {
822                return Err(HiveGpuError::VectorNotFound(id.clone()));
823            }
824        }
825        Ok(())
826    }
827
828    fn vector_count(&self) -> usize {
829        self.vector_count.saturating_sub(self.removed_indices.len())
830    }
831
832    fn dimension(&self) -> usize {
833        self.dimension
834    }
835
836    fn get_vector(&self, id: &str) -> Result<Option<GpuVector>> {
837        let Some(&index) = self.vector_id_map.get(id) else {
838            return Ok(None);
839        };
840        if self.removed_indices.contains(&index) {
841            return Ok(None);
842        }
843        let offset_bytes = index * self.dimension * std::mem::size_of::<f32>();
844        let storage = self.storage.as_ref().expect("storage live");
845        // Read straight from the mapped host pointer — same pattern as
846        // `read_f32_vec` but starting at an offset.
847        if storage.host_ptr.is_null() {
848            return Err(HiveGpuError::VulkanError(
849                "storage buffer not host-visible".to_string(),
850            ));
851        }
852        let mut host = vec![0f32; self.dimension];
853        // SAFETY: offset + dimension bounded by vector_count * dimension.
854        unsafe {
855            std::ptr::copy_nonoverlapping(
856                storage.host_ptr.add(offset_bytes) as *const u8,
857                host.as_mut_ptr() as *mut u8,
858                self.dimension * std::mem::size_of::<f32>(),
859            );
860        }
861        let metadata = self.payloads.get(id).cloned().unwrap_or_default();
862        Ok(Some(GpuVector {
863            id: id.to_string(),
864            data: host,
865            metadata,
866        }))
867    }
868
869    fn clear(&mut self) -> Result<()> {
870        self.vector_count = 0;
871        self.buffer_capacity = self.buffer_capacity.max(MIN_INITIAL_VECTORS);
872        self.vector_id_map.clear();
873        self.index_to_id.clear();
874        self.removed_indices.clear();
875        self.payloads.clear();
876        self.norms_sq.clear();
877        Ok(())
878    }
