1//! # CUDA Helpers
2//!
3//! Helper functions for CUDA GPU operations.

5use super::context::CudaContext;
6use crate::error::{HiveGpuError, Result};
7
8/// CUDA Helper Functions
9#[cfg(feature = "cuda")]
10pub struct CudaHelpers;
11
12#[cfg(feature = "cuda")]
13impl CudaHelpers {
14    /// Calculate optimal block size for CUDA kernels
15    pub fn calculate_block_size(
16        context: &CudaContext,
17        grid_size: (u32, u32, u32),
18    ) -> Result<(u32, u32, u32)> {
19        // Calculate optimal block size based on device capabilities
20        let (major, minor) = context.compute_capability();
21
22        // Block size limits based on compute capability
23        let max_threads_per_block = if major >= 7 {
24            1024 // Volta and newer
25        } else if major >= 6 {
26            1024 // Pascal
27        } else {
28            512 // Older architectures
29        };
30
31        // Calculate optimal block size
32        let x = grid_size.0.min(max_threads_per_block);
33        let y = grid_size.1.min(max_threads_per_block / x);
34        let z = grid_size.2.min(max_threads_per_block / (x * y));
35
36        Ok((x, y, z))
37    }
38
39    /// Validate CUDA device capabilities
40    pub fn validate_device_capabilities(context: &CudaContext) -> Result<()> {
41        // Check if device supports required features
42        if !context.supports_required_features() {
43            return Err(HiveGpuError::Other(
44                "CUDA device does not support required features".to_string(),
45            ));
46        }
47
48        Ok(())
49    }
50}
