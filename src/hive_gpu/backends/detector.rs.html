1//! GPU Backend Detection
2//!
3//! This module provides detection of available GPU backends and selection
4//! of the best backend for the current system.
5
6use crate::error::{HiveGpuError, Result};
7
8/// Available GPU backend types
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub enum GpuBackendType {
11    /// Metal (Apple Silicon)
12    Metal,
13    /// CUDA (NVIDIA)
14    Cuda,
15    /// CPU fallback
16    Cpu,
17}
18
19impl std::fmt::Display for GpuBackendType {
20    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
21        match self {
22            GpuBackendType::Metal => write!(f, "Metal"),
23            GpuBackendType::Cuda => write!(f, "CUDA"),
24            GpuBackendType::Cpu => write!(f, "CPU"),
25        }
26    }
27}
28
29/// Detect all available GPU backends on the current system
30pub fn detect_available_backends() -> Vec<GpuBackendType> {
31    let mut backends = Vec::new();
32
33    // Check Metal availability (macOS only)
34    #[cfg(all(target_os = "macos", feature = "metal-native"))]
35    {
36        if is_metal_available() {
37            backends.push(GpuBackendType::Metal);
38        }
39    }
40
41    // Check CUDA availability
42    #[cfg(feature = "cuda")]
43    {
44        if is_cuda_available() {
45            backends.push(GpuBackendType::Cuda);
46        }
47    }
48
49    // CPU is always available as fallback
50    backends.push(GpuBackendType::Cpu);
51
52    backends
53}
54
55/// Select the best available backend based on performance priority
56pub fn select_best_backend() -> Result<GpuBackendType> {
57    let available = detect_available_backends();
58
59    // Priority order: Metal > CUDA > CPU
60    if available.contains(&GpuBackendType::Metal) {
61        Ok(GpuBackendType::Metal)
62    } else if available.contains(&GpuBackendType::Cuda) {
63        Ok(GpuBackendType::Cuda)
64    } else if available.contains(&GpuBackendType::Cpu) {
65        Ok(GpuBackendType::Cpu)
66    } else {
67        Err(HiveGpuError::NoDeviceAvailable)
68    }
69}

71/// Check if Metal is available on the current system
72#[cfg(all(target_os = "macos", feature = "metal-native"))]
73fn is_metal_available() -> bool {
74    use objc2_metal::MTLCreateSystemDefaultDevice;
75
76    MTLCreateSystemDefaultDevice().is_some()
77}
78
79/// Check if CUDA is available on the current system
80#[cfg(feature = "cuda")]
81fn is_cuda_available() -> bool {
82    // This is a simplified check - in practice you'd check for CUDA runtime
83    // and available devices
84    std::env::var("CUDA_VISIBLE_DEVICES").is_ok() || std::env::var("CUDA_HOME").is_ok()
85}

87/// Get backend-specific device information
88pub fn get_backend_info(backend: GpuBackendType) -> Result<String> {
89    match backend {
90        GpuBackendType::Metal => {
91            #[cfg(all(target_os = "macos", feature = "metal-native"))]
92            {
93                use objc2_metal::{MTLCreateSystemDefaultDevice, MTLDevice};
94                if let Some(device) = MTLCreateSystemDefaultDevice() {
95                    Ok(format!("Metal device: {}", device.name()))
96                } else {
97                    Err(HiveGpuError::NoDeviceAvailable)
98                }
99            }
100            #[cfg(not(all(target_os = "macos", feature = "metal-native")))]
101            {
102                Err(HiveGpuError::NoDeviceAvailable)
103            }
104        }
105        GpuBackendType::Cuda => {
106            #[cfg(feature = "cuda")]
107            {
108                // In practice, you'd query CUDA devices here
109                Ok("CUDA device available".to_string())
110            }
111            #[cfg(not(feature = "cuda"))]
112            {
113                Err(HiveGpuError::NoDeviceAvailable)
114            }
115        }
116        GpuBackendType::Cpu => Ok("CPU fallback".to_string()),
117    }
118}
