2//!
3//! Real context backed by `libamdhip64` + `librocblas` loaded dynamically
4//! via [`super::ffi::hip_lib`]. Mirrors the shape of

26/// ROCm driver context + rocBLAS handle for a single GPU ordinal.
27pub struct RocmContext {
28    device_id: i32,
29    stream: HipStream_t,
30    rocblas_handle: RocblasHandle,
31}
32
33// SAFETY: All three fields are opaque device-side handles. Access is
34// serialised via the owning stream; we never read the raw pointers from
35// multiple threads without going through rocBLAS/HIP (which internally
36// synchronise on the stream).
37unsafe impl Send for RocmContext {}
38unsafe impl Sync for RocmContext {}

52    pub fn new() -> Result<Arc<Self>> {
53        Self::new_with_device(0)
54    }

56    /// Create a context on a specific device ordinal.
57    pub fn new_with_device(ordinal: i32) -> Result<Arc<Self>> {
58        let lib = ffi::require_hip_lib()?;
59
60        // `hipInit(0)` is idempotent — the HIP runtime tolerates repeat
61        // calls. Safe to call on every context creation.
62        // SAFETY: `hip_init` has no preconditions beyond the library
63        // being loaded.
64        let status = unsafe { (lib.hip_init)(0) };
65        hip_check(status, "hipInit")?;
66
67        // SAFETY: `hip_set_device` takes a device ordinal; we validate
68        // bounds via `hipGetDeviceCount` implicitly (HIP returns an error
69        // status if the ordinal is out of range).
70        let status = unsafe { (lib.hip_set_device)(ordinal) };
71        hip_check(status, "hipSetDevice")?;
72
73        // Create a stream dedicated to this context. The stream owns all
74        // operations dispatched through rocBLAS and HIP memcpy calls.
75        let mut stream: HipStream_t = std::ptr::null_mut();
76        // SAFETY: We pass a valid out-pointer; HIP writes the newly
77        // created stream handle into it on success.
78        let status = unsafe { (lib.hip_stream_create)(&mut stream) };
79        hip_check(status, "hipStreamCreate")?;
80
81        // rocBLAS handle bound to the stream so every SGEMV/SGEMM runs on
82        // our serialised queue.
83        let mut rocblas_handle: RocblasHandle = std::ptr::null_mut();
84        // SAFETY: `rocblas_create_handle` writes the handle via the
85        // out-pointer; callers must pair every successful create with a
86        // `rocblas_destroy_handle`. Our `Drop` impl does exactly that.
87        let status = unsafe { (lib.rocblas_create_handle)(&mut rocblas_handle) };
88        if let Err(e) = rocblas_check(status, "rocblas_create_handle") {
89            // Clean up the HIP stream we already created.
90            // SAFETY: `stream` is a valid handle we just created; HIP
91            // tolerates destroying a stream with queued work.
92            unsafe {
93                let _ = (lib.hip_stream_destroy)(stream);
94            }
95            return Err(e);
96        }
97
98        // Attach the stream to rocBLAS so every subsequent rocBLAS call
99        // queues on the same stream as the HIP memcpy operations.
100        // SAFETY: both handles were just created successfully.
101        let status = unsafe { (lib.rocblas_set_stream)(rocblas_handle, stream) };
102        if let Err(e) = rocblas_check(status, "rocblas_set_stream") {
103            unsafe {
104                let _ = (lib.rocblas_destroy_handle)(rocblas_handle);
105                let _ = (lib.hip_stream_destroy)(stream);
106            }
107            return Err(e);
108        }
109
110        debug!("rocm context ready: ordinal={}", ordinal);
111        Ok(Arc::new(Self {
112            device_id: ordinal,
113            stream,
114            rocblas_handle,
115        }))
116    }

118    /// Non-failing availability probe used by the backend detector.
119    pub fn is_available() -> bool {
120        let Some(lib) = ffi::hip_lib() else {
121            return false;
122        };
123        // SAFETY: the HIP runtime is safe to query for device count even
124        // before `hipInit`; if the driver is not loaded the call returns
125        // a non-zero status.
126        let mut count: c_int = 0;
127        let init_status = unsafe { (lib.hip_init)(0) };
128        if init_status != 0 {
129            return false;
130        }
131        let status = unsafe { (lib.hip_get_device_count)(&mut count) };
132        status == 0 && count > 0
133    }
134
135    /// Count HIP devices on the host. Returns 0 when the runtime is
136    /// absent.
137    pub fn device_count() -> Result<usize> {
138        let lib = ffi::require_hip_lib()?;
139        // SAFETY: `hip_init` is idempotent; `hip_get_device_count` writes
140        // a single int to the provided pointer.
141        let mut count: c_int = 0;
142        let status = unsafe { (lib.hip_init)(0) };
143        hip_check(status, "hipInit")?;
144        let status = unsafe { (lib.hip_get_device_count)(&mut count) };
145        hip_check(status, "hipGetDeviceCount")?;
146        Ok(count.max(0) as usize)
147    }

165    /// Human-readable device name from `hipDeviceGetName`.
166    pub fn device_name(&self) -> String {
167        let Ok(lib) = ffi::require_hip_lib() else {
168            return "unknown".to_string();
169        };
170        let mut buf = [0i8; 256];
171        // SAFETY: HIP writes a null-terminated UTF-8 string into `buf`;
172        // we cap the length so there is no overflow risk.
173        let status = unsafe {
174            (lib.hip_device_get_name)(
175                buf.as_mut_ptr() as *mut c_char,
176                buf.len() as c_int,
177                self.device_id,
178            )
179        };
180        if status != 0 {
181            return "unknown".to_string();
182        }
183        // SAFETY: buffer is null-terminated on success.
184        unsafe {
185            let bytes: &[u8] = std::slice::from_raw_parts(
186                buf.as_ptr() as *const u8,
187                buf.iter().position(|&c| c == 0).unwrap_or(buf.len()),
188            );
189            String::from_utf8_lossy(bytes).to_string()
190        }
191    }

193    /// gfx architecture inferred from compute capability attributes.
194    /// ROCm exposes `MAJOR` / `MINOR` that map to the gfx codes most
195    /// users recognise (e.g. (10, 3) → gfx1030).
196    pub fn compute_capability(&self) -> (u32, u32) {
197        let Ok(lib) = ffi::require_hip_lib() else {
198            return (0, 0);
199        };
200        let major = get_attribute(
201            lib,
202            HIP_DEVICE_ATTR_COMPUTE_CAPABILITY_MAJOR,
203            self.device_id,
204        );
205        let minor = get_attribute(
206            lib,
207            HIP_DEVICE_ATTR_COMPUTE_CAPABILITY_MINOR,
208            self.device_id,
209        );
210        (major.max(0) as u32, minor.max(0) as u32)
211    }
212
213    /// Format the `gfx<major><minor>` string that ROCm ecosystems use.
214    pub fn gfx_string(&self) -> String {
215        let (maj, min) = self.compute_capability();
216        format!("gfx{maj}{min:02}")
217    }

219    /// Total physical VRAM in bytes.
220    pub fn total_memory(&self) -> u64 {
221        Self::mem_info(self.device_id)
222            .map(|(_, total)| total)
223            .unwrap_or(0)
224    }

226    /// Free VRAM in bytes, queried live.
227    pub fn available_memory(&self) -> u64 {
228        Self::mem_info(self.device_id)
229            .map(|(free, _)| free)
230            .unwrap_or(0)
231    }
232
233    fn mem_info(device: HipDevice_t) -> Result<(u64, u64)> {
234        let lib = ffi::require_hip_lib()?;
235        // SAFETY: `hip_set_device` is safe; `hip_mem_get_info` writes two
236        // usize values via out-pointers.
237        let status = unsafe { (lib.hip_set_device)(device) };
238        hip_check(status, "hipSetDevice")?;
239        let mut free: usize = 0;
240        let mut total: usize = 0;
241        let status = unsafe { (lib.hip_mem_get_info)(&mut free, &mut total) };
242        hip_check(status, "hipMemGetInfo")?;
243        Ok((free as u64, total as u64))
244    }
245
246    fn pci_bus_id_string(&self) -> Option<String> {
247        let lib = ffi::require_hip_lib().ok()?;
248        let bus = get_attribute(lib, HIP_DEVICE_ATTR_PCI_BUS_ID, self.device_id);
249        let dev = get_attribute(lib, HIP_DEVICE_ATTR_PCI_DEVICE_ID, self.device_id);
250        let domain = get_attribute(lib, HIP_DEVICE_ATTR_PCI_DOMAIN_ID, self.device_id);
251        if bus < 0 || dev < 0 {
252            return None;
253        }
254        Some(format!("{domain:04x}:{bus:02x}:{dev:02x}.0"))
255    }
256
257    fn driver_version_string() -> String {
258        let Ok(lib) = ffi::require_hip_lib() else {
259            return "ROCm unknown".to_string();
260        };
261        let mut version: c_int = 0;
262        // SAFETY: `hipDriverGetVersion` is safe; it writes an int.
263        let status = unsafe { (lib.hip_driver_get_version)(&mut version) };
264        if status != 0 {
265            return "ROCm unknown".to_string();
266        }
267        format!("ROCm {}.{}", version / 1000, (version % 1000) / 10)
268    }
269}

271fn get_attribute(lib: &ffi::HipLib, attr: c_int, device: HipDevice_t) -> c_int {
272    let mut value: c_int = 0;
273    // SAFETY: `hipDeviceGetAttribute` writes a single int.
274    let status = unsafe { (lib.hip_device_get_attribute)(&mut value, attr, device) };
275    if status == 0 { value } else { -1 }
276}

278impl Drop for RocmContext {
279    fn drop(&mut self) {
280        let Some(lib) = ffi::hip_lib() else {
281            return;
282        };
283        // Destruction order matters: drop rocBLAS first so it can sync
284        // any pending work before the stream goes away. Swallow errors
285        // on teardown — there is nothing useful a caller can do.
286        // SAFETY: both handles were created by us in `new_with_device`
287        // and are not borrowed elsewhere at this point (we hold a
288        // unique `&mut self`).
289        if !self.rocblas_handle.is_null() {
290            unsafe {
291                let _ = (lib.rocblas_destroy_handle)(self.rocblas_handle);
292            }
293            self.rocblas_handle = std::ptr::null_mut();
294        }
295        if !self.stream.is_null() {
296            unsafe {
297                let _ = (lib.hip_stream_destroy)(self.stream);
298            }
299            self.stream = std::ptr::null_mut();
300        }
301    }
302}

304impl GpuBackend for RocmContext {
305    fn device_info(&self) -> GpuDeviceInfo {
306        let (free, total) = Self::mem_info(self.device_id).unwrap_or((0, 0));
307        let used = total.saturating_sub(free);
308        let max_threads_per_block = ffi::hip_lib()
309            .map(|l| get_attribute(l, HIP_DEVICE_ATTR_MAX_THREADS_PER_BLOCK, self.device_id))
310            .unwrap_or(0) as u32;
311        let max_shared_memory_per_block = ffi::hip_lib()
312            .map(|l| {
313                get_attribute(
314                    l,
315                    HIP_DEVICE_ATTR_MAX_SHARED_MEMORY_PER_BLOCK,
316                    self.device_id,
317                )
318            })
319            .unwrap_or(0) as u64;
320        GpuDeviceInfo {
321            name: self.device_name(),
322            backend: "ROCm".to_string(),
323            total_vram_bytes: total,
324            available_vram_bytes: free,
325            used_vram_bytes: used,
326            driver_version: Self::driver_version_string(),
327            compute_capability: Some(self.gfx_string()),
328            max_threads_per_block,
329            max_shared_memory_per_block,
330            device_id: self.device_id,
331            pci_bus_id: self.pci_bus_id_string(),
332        }
333    }
334
335    fn supports_operations(&self) -> GpuCapabilities {
336        GpuCapabilities {
337            supports_hnsw: false,
338            supports_batch: true,
339            max_dimension: 4096,
340            max_batch_size: 100_000,
341        }
342    }
343
344    fn memory_stats(&self) -> GpuMemoryStats {
345        let (free, total) = Self::mem_info(self.device_id).unwrap_or((0, 0));
346        let used = total.saturating_sub(free) as usize;
347        let available = free as usize;
348        let utilization = if total == 0 {
349            0.0
350        } else {
351            used as f32 / total as f32
352        };
353        GpuMemoryStats {
354            total_allocated: used,
355            available,
356            utilization,
357            buffer_count: 0,
358        }
359    }
360}

362impl GpuContext for RocmContext {
363    fn create_storage(
364        &self,
365        dimension: usize,
366        metric: crate::types::GpuDistanceMetric,
367    ) -> Result<Box<dyn crate::traits::GpuVectorStorage>> {
368        use super::vector_storage::RocmVectorStorage;
369        // The vector storage needs a cloned Arc<Self>. Since `GpuContext`
370        // methods take `&self`, we materialise a fresh Arc view that
371        // aliases the existing handles. Multiple `Arc<RocmContext>` are
372        // safe because `Drop` is tied to the last strong ref.
373        let storage = RocmVectorStorage::new(
374            // SAFETY: the caller must guarantee `self` is already held in
375            // an `Arc`. The `create_storage` trait contract only makes
376            // sense when the context is owned that way — the public
377            // constructor always returns `Arc<Self>`.
378            unsafe {
379                let raw = self as *const Self;
380                Arc::increment_strong_count(raw);
381                Arc::from_raw(raw)
382            },
383            dimension,
384            metric,
385        )?;
386        Ok(Box::new(storage))
387    }
388
389    fn create_storage_with_config(
390        &self,
391        dimension: usize,
392        metric: crate::types::GpuDistanceMetric,
393        _config: crate::types::HnswConfig,
394    ) -> Result<Box<dyn crate::traits::GpuVectorStorage>> {
395        self.create_storage(dimension, metric)
396    }
397
398    fn memory_stats(&self) -> GpuMemoryStats {
399        GpuBackend::memory_stats(self)
400    }
401
402    fn device_info(&self) -> Result<GpuDeviceInfo> {
403        Ok(GpuBackend::device_info(self))
404    }
